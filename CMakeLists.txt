cmake_minimum_required(VERSION 3.25)
project(windows_lhdc VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
set(LHDCV5_BUILD_TESTS "${BUILD_TESTING}")
add_subdirectory(third_party/lhdcv5)
add_library(lhdc_core
    src/control/hey_melody.cpp
    src/avrcp/protocol.cpp
    src/avdtp/protocol.cpp
    src/avdtp/lhdc_v5.cpp
    src/avdtp/stream.cpp
    src/codec/encoder.cpp
    src/codec/bitrate_control.cpp
    src/codec/profile.cpp
    src/media/packetizer.cpp
    src/audio/wav.cpp
    src/audio/pcm_queue.cpp)
target_include_directories(lhdc_core PUBLIC src include)
target_link_libraries(lhdc_core PUBLIC lhdcv5_enc)
if(MSVC)
    target_compile_options(lhdc_core PRIVATE /W4 /WX /utf-8)
    target_compile_options(lhdcv5_enc PRIVATE /fp:precise)
endif()
add_executable(lhdc-host src/host/main.cpp)
target_link_libraries(lhdc-host PRIVATE lhdc_core)
if(WIN32)
    add_library(lhdc_windows src/control/rfcomm.cpp src/transport/windows.cpp src/host/signalling.cpp src/host/pacer.cpp src/host/media_session.cpp src/audio/direct_pcm.cpp src/audio/pcm_converter.cpp src/audio/pcm_reader.cpp src/audio/volume_sync.cpp src/audio/hfp_activity.cpp src/service/status.cpp)
    target_link_libraries(lhdc_windows PUBLIC lhdc_core setupapi cfgmgr32 Bthprops ole32 avrt uuid advapi32 ws2_32)
    target_compile_definitions(lhdc_windows PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_sources(lhdc-host PRIVATE src/audio/wasapi.cpp)
    target_link_libraries(lhdc-host PRIVATE lhdc_windows)
    add_executable(lhdc-service src/service/main.cpp src/service/stream.cpp src/service/worker.cpp)
    target_link_libraries(lhdc-service PRIVATE lhdc_windows)
    add_executable(lhdc-device installer/device_tool.cpp installer/signature.cpp)
    target_include_directories(lhdc-device PRIVATE src)
    target_link_libraries(lhdc-device PRIVATE lhdc_windows newdev ntdll crypt32 wintrust)
    add_executable(lhdc-audio-device installer/audio_device.cpp installer/signature.cpp)
    target_include_directories(lhdc-audio-device PRIVATE src include)
    target_link_libraries(lhdc-audio-device PRIVATE lhdc_windows newdev crypt32 wintrust)
    if(MSVC)
        target_compile_options(lhdc_windows PRIVATE /W4 /WX /utf-8)
        target_compile_options(lhdc-service PRIVATE /W4 /WX /utf-8)
        target_compile_options(lhdc-device PRIVATE /W4 /WX /utf-8)
        target_compile_options(lhdc-audio-device PRIVATE /W4 /WX /utf-8)
    endif()
endif()
if(MSVC)
    target_compile_options(lhdc-host PRIVATE /W4 /WX /utf-8)
endif()
if(BUILD_TESTING)
    add_executable(lhdc-control-tests tests/control_tests.cpp)
    target_link_libraries(lhdc-control-tests PRIVATE lhdc_core)
    add_test(NAME earbud_control COMMAND lhdc-control-tests)
    add_executable(lhdc-avrcp-tests tests/avrcp_tests.cpp)
    target_link_libraries(lhdc-avrcp-tests PRIVATE lhdc_core)
    add_test(NAME avrcp COMMAND lhdc-avrcp-tests)
    add_executable(lhdc-progress-log-tests tests/progress_log_tests.cpp)
    target_include_directories(lhdc-progress-log-tests PRIVATE src)
    add_test(NAME progress_log COMMAND lhdc-progress-log-tests)
    add_executable(lhdc-tests tests/core_tests.cpp)
    target_link_libraries(lhdc-tests PRIVATE lhdc_core)
    add_test(NAME core COMMAND lhdc-tests)
    add_executable(lhdc-audio-tests tests/audio_tests.cpp)
    target_link_libraries(lhdc-audio-tests PRIVATE lhdc_core)
    add_test(NAME audio COMMAND lhdc-audio-tests)
    if(WIN32)
        add_executable(lhdc-windows-audio-tests tests/windows_audio_tests.cpp)
        target_link_libraries(lhdc-windows-audio-tests PRIVATE lhdc_windows)
        if(MSVC)
            target_compile_options(lhdc-windows-audio-tests PRIVATE /W4 /WX /utf-8)
        endif()
        add_test(NAME windows_audio COMMAND lhdc-windows-audio-tests)
        add_executable(lhdc-pcm-reader-tests tests/pcm_reader_tests.cpp)
        target_link_libraries(lhdc-pcm-reader-tests PRIVATE lhdc_windows)
        if(MSVC)
            target_compile_options(lhdc-pcm-reader-tests PRIVATE /W4 /WX /utf-8)
        endif()
        add_test(NAME pcm_reader COMMAND lhdc-pcm-reader-tests)
        find_program(POWERSHELL_EXECUTABLE pwsh REQUIRED)
        add_test(NAME playback_assessment COMMAND "${POWERSHELL_EXECUTABLE}" -NoProfile -File "${CMAKE_SOURCE_DIR}/tests/long_playback_tests.ps1")
    endif()
endif()
